/* ================================================================
   Shine.com – Universal Job Finder  (interactive form)
   ----------------------------------------------------------------
   Paste into the DevTools Console on a Shine search page, e.g.
   shine.com/job-search/android-developer-jobs. A form appears — set
   Keywords / Exclude, click ▶ Start. CSV downloads when done.

   HOW IT WORKS
   • Shine pages are server-rendered and fetchable, so the tool pulls
     each results page in the background (page URLs end in -2, -3, …)
     and parses the cards — no navigating you around, no API/CAPTCHA.
   • Each card gives title, company, experience, salary, location and
     skills. Keywords match against title + skills + company.
   • Change "Profile" to jump to a different role's search (a reload —
     re-paste & Start there).

   HOW TO USE
   1. Open shine.com/job-search/<role>-jobs (e.g. android-developer-jobs).
   2. Console (Mac: Cmd+Option+J | Win: Ctrl+Shift+J) → paste → Enter.
   3. Fill the form → ▶ Start.
   ================================================================ */

(function () {
  const $ = id => document.getElementById(id);
  const esc = s => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const slugify = s => s.trim().toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
  const delay = ms => new Promise(r => setTimeout(r, ms));

  const mk = (tag, css, txt) => { const e = document.createElement(tag); if (css) e.style.cssText = css; if (txt != null) e.textContent = txt; return e; };
  function inputEl(id, val, ph) { const e = document.createElement("input"); e.id = id; e.value = val || ""; e.placeholder = ph || ""; e.style.cssText = "width:100%;box-sizing:border-box;padding:6px 8px;border:1px solid #f5a623;border-radius:6px;background:#241c07;color:#fff6e5;font-size:12px"; return e; }
  function fieldBlock(label, id, val, ph) { const w = mk("div"); w.appendChild(mk("label", "display:block;margin:8px 0 3px;font-size:11px;opacity:.85", label)); w.appendChild(inputEl(id, val, ph)); return w; }
  function checkBlock(label, id, on) { const l = mk("label", "display:flex;align-items:center;gap:6px;margin:6px 0;font-size:12px;cursor:pointer"); const c = document.createElement("input"); c.type = "checkbox"; c.id = id; c.checked = !!on; l.appendChild(c); l.appendChild(document.createTextNode(" " + label)); return l; }

  const roleMatch = location.pathname.match(/\/job-search\/(.+?)-jobs(?:-\d+)?$/);
  const curRole = roleMatch ? roleMatch[1].replace(/-/g, " ") : "";

  $("shf-panel")?.remove();
  const p = mk("div", "position:fixed;top:16px;right:16px;width:340px;max-height:88vh;z-index:2147483647;background:#1a1405;color:#fff6e5;font:13px/1.45 -apple-system,Segoe UI,Roboto,sans-serif;border:1px solid #f5a623;border-radius:10px;box-shadow:0 8px 30px rgba(0,0,0,.5);display:flex;flex-direction:column;overflow:hidden");
  p.id = "shf-panel";
  const header = mk("div", "padding:10px 12px;background:#8a5a00;font-weight:600;display:flex;justify-content:space-between;align-items:center");
  header.appendChild(mk("span", "", "☀ Shine Job Finder"));
  const closeX = mk("span", "cursor:pointer;opacity:.7", "✕"); header.appendChild(closeX); p.appendChild(header);

  const form = mk("div", "padding:10px 12px;overflow:auto"); form.id = "shf-form";
  form.appendChild(fieldBlock("Profile / role (change = go to that page)", "shf-profile", curRole, "e.g. android developer"));
  form.appendChild(fieldBlock("Keywords (comma-sep — ANY match; blank = all)", "shf-keywords", "android, kotlin, jetpack, flutter", "android, kotlin"));
  form.appendChild(fieldBlock("Exclude words (comma-sep — optional)", "shf-exclude", "", "senior, sales"));
  const two = mk("div", "display:flex;gap:8px");
  const c1 = mk("div", "flex:1"); c1.appendChild(fieldBlock("Max pages", "shf-maxpages", "50"));
  const c2 = mk("div", "flex:1"); c2.appendChild(fieldBlock("Stop after N (0=all)", "shf-stop", "0"));
  two.appendChild(c1); two.appendChild(c2); form.appendChild(two);
  const opts = mk("div", "margin-top:8px;padding-top:6px;border-top:1px solid #f5a623");
  opts.appendChild(checkBlock("Open each match in a new tab", "shf-opentabs", false));
  form.appendChild(opts);
  const startBtn = mk("button", "width:100%;margin-top:12px;padding:9px;border:0;border-radius:7px;background:#f5a623;color:#1a1405;font-weight:700;font-size:13px;cursor:pointer", "▶ Start search");
  startBtn.id = "shf-start"; form.appendChild(startBtn); p.appendChild(form);
  const status = mk("div", "padding:8px 12px;border-top:1px solid #f5a623;font-size:12px;display:none"); status.id = "shf-status";
  const list = mk("div", "overflow:auto;padding:2px 0"); list.id = "shf-list";
  p.appendChild(status); p.appendChild(list);
  document.body.appendChild(p);
  closeX.onclick = () => p.remove();
  startBtn.onclick = start;

  const say = m => { status.textContent = m; console.log(m); };
  function addRow(r) {
    const row = mk("div", "padding:6px 12px;border-bottom:1px solid #2e2410");
    const a = mk("a", "color:#ffcf7a;text-decoration:none;font-weight:600;display:block", r.title);
    a.href = r.url; a.target = "_blank"; row.appendChild(a);
    row.appendChild(mk("div", "font-size:11px;opacity:.8", [r.company, r.location, r.salary].filter(x => x && x !== "—").join(" · ")));
    list.appendChild(row); list.scrollTop = list.scrollHeight;
  }

  const txt = (el, sel) => { const n = el.querySelector(sel); return n ? (n.textContent || "").trim() : ""; };

  function extractCard(card) {
    const title = txt(card, '[class*="bigCardTopTitleHeading"]');
    const company = txt(card, '[class*="bigCardTopTitleName"]');
    const exps = [...card.querySelectorAll('[class*="bigCardExperience"]')].map(e => (e.textContent || "").trim());
    const exp = exps.find(t => /Yrs?/i.test(t)) || "";
    const salary = exps.find(t => /LPA|Lacs|₹/i.test(t)) || "—";
    const location = txt(card, '[class*="bigCardLocation"]');
    const skills = [...card.querySelectorAll('[class*="Skill"] span, [class*="skill"] a, [class*="Tag"] span')].map(e => (e.textContent || "").trim()).filter(Boolean).slice(0, 10).join(", ");
    const href = card.querySelector('a[href]')?.getAttribute("href") || "";
    const url = href ? (href.startsWith("http") ? href : "https://www.shine.com" + href) : "";
    return { title, company, exp, salary, location, skills, url };
  }

  async function start() {
    const splitList = s => s.split(",").map(x => x.trim()).filter(Boolean);
    const profile = $("shf-profile").value.trim();
    if (profile && slugify(profile) !== slugify(curRole)) {
      say(`↪ Switching to "${profile}" — reloads; re-paste & Start.`);
      await delay(1000);
      location.href = "https://www.shine.com/job-search/" + slugify(profile) + "-jobs";
      return;
    }
    const cfg = {
      KEYWORDS: splitList($("shf-keywords").value),
      EXCLUDE:  splitList($("shf-exclude").value),
      MAX_PAGES:Math.max(1, parseInt($("shf-maxpages").value) || 50),
      STOP_AT:  Math.max(0, parseInt($("shf-stop").value) || 0),
      OPEN_TABS:$("shf-opentabs").checked,
      DELAY_MS: 800,
    };
    form.style.display = "none"; status.style.display = "block";
    await runSearch(cfg);
  }

  async function runSearch(cfg) {
    const kwRx = cfg.KEYWORDS.length ? new RegExp(cfg.KEYWORDS.map(esc).join("|"), "i") : null;
    const exRx = cfg.EXCLUDE.length  ? new RegExp(cfg.EXCLUDE.map(esc).join("|"),  "i") : null;
    const basePath = "https://www.shine.com/job-search/" + slugify(curRole || "android-developer") + "-jobs";

    say("Scanning Shine results...");
    const seen = new Set(), matches = [];

    for (let page = 1; page <= cfg.MAX_PAGES; page++) {
      const url = page === 1 ? basePath : basePath + "-" + page;
      let doc;
      try {
        const r = await fetch(url, { credentials: "include" });
        if (r.status !== 200) { say(`⚠ page ${page}: HTTP ${r.status} — stopping.`); break; }
        doc = new DOMParser().parseFromString(await r.text(), "text/html");
      } catch (e) { say(`⚠ page ${page} failed — stopping.`); break; }

      const cards = [...doc.querySelectorAll('[class*="jobCardNova_bigCard__"]')];
      if (!cards.length) { say(`📄 Page ${page}: no jobs — reached the end.`); break; }

      let hits = 0;
      for (const card of cards) {
        let j; try { j = extractCard(card); } catch { continue; }
        if (!j.title || !j.url || seen.has(j.url)) continue;
        seen.add(j.url);
        const text = [j.title, j.skills, j.company].join(" ");
        if (kwRx && !kwRx.test(text)) continue;
        if (exRx && exRx.test(text)) continue;
        matches.push(j); addRow(j); hits++;
        console.log(`✓ ${j.title} — ${j.company} — ${j.url}`);
      }
      say(`page ${page}: ${cards.length} jobs, ${hits} match. Total: ${matches.length}`);
      if (cfg.STOP_AT && matches.length >= cfg.STOP_AT) { say(`🎯 Reached ${cfg.STOP_AT} — stopping.`); break; }
      await delay(cfg.DELAY_MS);
    }

    if (!matches.length) { say("No matches found (try broader keywords)."); return; }

    console.table(matches.map(({ title, company, location, exp, salary }) => ({ title, company, location, exp, salary })));
    const q = s => `"${String(s).replace(/"/g, '""')}"`;
    const csv = "Title,Company,Location,Experience,Salary,Skills,URL\n" +
      matches.map(m => [q(m.title), q(m.company), q(m.location), q(m.exp), q(m.salary), q(m.skills), q(m.url)].join(",")).join("\n");
    const a = mk("a"); a.href = URL.createObjectURL(new Blob([csv], { type: "text/csv" }));
    a.download = `shine-jobs-${slugify(curRole || "search")}.csv`;
    document.body.appendChild(a); a.click(); a.remove();
    say(`✅ Done — ${matches.length} match(es). Saved ${a.download}.`);
    if (cfg.OPEN_TABS) for (const m of matches) { window.open(m.url, "_blank"); await delay(500); }
  }
})();
