/* ================================================================
   Instahyre – Universal Job Finder  (interactive form)
   ----------------------------------------------------------------
   Paste into the DevTools Console while logged in on Instahyre
   (instahyre.com/search-jobs). A form appears — set Skill/Profile,
   Keywords, Exclude, then ▶ Start. CSV downloads when done.

   HOW IT WORKS
   • Instahyre exposes a clean JSON API (api/v1/job_search) that uses
     your logged-in session — no CAPTCHA. The tool queries it with
     your skill, pages through results via `offset`, and collects
     title, company, locations, skills and the job URL.
   • "Skill / Profile" maps to Instahyre's skill filter (e.g. Android).
     "Keywords" further narrows client-side by title + skills.

   HOW TO USE
   1. Log in to Instahyre.
   2. Console (Mac: Cmd+Option+J | Win: Ctrl+Shift+J) → paste → Enter.
   3. Set Skill (e.g. Android) → ▶ Start.

   NOTE: keep DELAY sane; the API is session-auth'd but still rate-limited.
   ================================================================ */

(function () {
  const $ = id => document.getElementById(id);
  const esc = s => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const delay = ms => new Promise(r => setTimeout(r, ms));

  const mk = (tag, css, txt) => { const e = document.createElement(tag); if (css) e.style.cssText = css; if (txt != null) e.textContent = txt; return e; };
  function inputEl(id, val, ph) { const e = document.createElement("input"); e.id = id; e.value = val || ""; e.placeholder = ph || ""; e.style.cssText = "width:100%;box-sizing:border-box;padding:6px 8px;border:1px solid #17b5a3;border-radius:6px;background:#0b1f1d;color:#eaf7f5;font-size:12px"; return e; }
  function fieldBlock(label, id, val, ph) { const w = mk("div"); w.appendChild(mk("label", "display:block;margin:8px 0 3px;font-size:11px;opacity:.85", label)); w.appendChild(inputEl(id, val, ph)); return w; }
  function checkBlock(label, id, on) { const l = mk("label", "display:flex;align-items:center;gap:6px;margin:6px 0;font-size:12px;cursor:pointer"); const c = document.createElement("input"); c.type = "checkbox"; c.id = id; c.checked = !!on; l.appendChild(c); l.appendChild(document.createTextNode(" " + label)); return l; }

  $("ihf-panel")?.remove();
  const p = mk("div", "position:fixed;top:16px;right:16px;width:340px;max-height:88vh;z-index:2147483647;background:#07211f;color:#eaf7f5;font:13px/1.45 -apple-system,Segoe UI,Roboto,sans-serif;border:1px solid #17b5a3;border-radius:10px;box-shadow:0 8px 30px rgba(0,0,0,.5);display:flex;flex-direction:column;overflow:hidden");
  p.id = "ihf-panel";
  const header = mk("div", "padding:10px 12px;background:#0d5a52;font-weight:600;display:flex;justify-content:space-between;align-items:center");
  header.appendChild(mk("span", "", "⚡ Instahyre Job Finder"));
  const closeX = mk("span", "cursor:pointer;opacity:.7", "✕"); header.appendChild(closeX); p.appendChild(header);

  const form = mk("div", "padding:10px 12px;overflow:auto"); form.id = "ihf-form";
  form.appendChild(fieldBlock("Skill / Profile (Instahyre skill filter)", "ihf-skill", "Android", "e.g. Android, React, Data Science"));
  form.appendChild(fieldBlock("Keywords (extra title/skill filter; blank = all)", "ihf-keywords", "", "android, kotlin, flutter"));
  form.appendChild(fieldBlock("Exclude words (comma-sep — optional)", "ihf-exclude", "", "senior, intern"));
  const two = mk("div", "display:flex;gap:8px");
  const c1 = mk("div", "flex:1"); c1.appendChild(fieldBlock("Max jobs", "ihf-maxjobs", "300"));
  const c2 = mk("div", "flex:1"); c2.appendChild(fieldBlock("Stop after N (0=all)", "ihf-stop", "0"));
  two.appendChild(c1); two.appendChild(c2); form.appendChild(two);
  const opts = mk("div", "margin-top:8px;padding-top:6px;border-top:1px solid #17b5a3");
  opts.appendChild(checkBlock("Open each match in a new tab", "ihf-opentabs", false));
  form.appendChild(opts);
  const startBtn = mk("button", "width:100%;margin-top:12px;padding:9px;border:0;border-radius:7px;background:#17b5a3;color:#00201d;font-weight:700;font-size:13px;cursor:pointer", "▶ Start search");
  startBtn.id = "ihf-start"; form.appendChild(startBtn); p.appendChild(form);
  const status = mk("div", "padding:8px 12px;border-top:1px solid #17b5a3;font-size:12px;display:none"); status.id = "ihf-status";
  const list = mk("div", "overflow:auto;padding:2px 0"); list.id = "ihf-list";
  p.appendChild(status); p.appendChild(list);
  document.body.appendChild(p);
  closeX.onclick = () => p.remove();
  startBtn.onclick = start;

  const say = m => { status.textContent = m; console.log(m); };
  function addRow(r) {
    const row = mk("div", "padding:6px 12px;border-bottom:1px solid #123");
    const a = mk("a", "color:#5fe6d3;text-decoration:none;font-weight:600;display:block", r.title);
    a.href = r.url; a.target = "_blank"; row.appendChild(a);
    row.appendChild(mk("div", "font-size:11px;opacity:.8", [r.company, r.locations].filter(Boolean).join(" · ")));
    list.appendChild(row); list.scrollTop = list.scrollHeight;
  }

  async function start() {
    const splitList = s => s.split(",").map(x => x.trim()).filter(Boolean);
    const cfg = {
      SKILL:    $("ihf-skill").value.trim(),
      KEYWORDS: splitList($("ihf-keywords").value),
      EXCLUDE:  splitList($("ihf-exclude").value),
      MAX_JOBS: Math.max(20, parseInt($("ihf-maxjobs").value) || 300),
      STOP_AT:  Math.max(0, parseInt($("ihf-stop").value) || 0),
      OPEN_TABS:$("ihf-opentabs").checked,
      DELAY_MS: 700,
    };
    form.style.display = "none"; status.style.display = "block";
    await runSearch(cfg);
  }

  async function runSearch(cfg) {
    const kwRx = cfg.KEYWORDS.length ? new RegExp(cfg.KEYWORDS.map(esc).join("|"), "i") : null;
    const exRx = cfg.EXCLUDE.length  ? new RegExp(cfg.EXCLUDE.map(esc).join("|"),  "i") : null;
    const base = "https://www.instahyre.com/api/v1/job_search?company_size=0&job_type=0&source=opportunities" +
      (cfg.SKILL ? "&skills=" + encodeURIComponent(cfg.SKILL) : "");

    say(`Searching Instahyre for "${cfg.SKILL || "all"}"...`);
    const seen = new Set(), matches = [];
    let total = Infinity;

    for (let offset = 0; offset < Math.min(cfg.MAX_JOBS, total); offset += 35) {
      let data;
      try {
        const r = await fetch(base + "&offset=" + offset, { credentials: "include", headers: { Accept: "application/json" } });
        if (r.status !== 200) { say(`⚠ offset ${offset}: HTTP ${r.status} — stopping.`); break; }
        data = await r.json();
      } catch (e) { say(`⚠ offset ${offset} failed — stopping.`); break; }

      total = data.meta?.total_count ?? total;
      const objs = data.objects || [];
      if (!objs.length) { say(`✅ Reached the end (${matches.length} kept).`); break; }

      for (const o of objs) {
        const url = o.public_url || "";
        if (!url || seen.has(url)) continue;
        seen.add(url);
        const title = o.title || "";
        const company = o.employer?.company_name || "";
        const locations = o.locations || "";
        const skills = (o.keywords || []).join(", ");
        const text = [title, skills].join(" ");
        if (kwRx && !kwRx.test(text)) continue;
        if (exRx && exRx.test(text)) continue;
        const rec = { title, company, locations, skills, url };
        matches.push(rec); addRow(rec);
        console.log(`✓ ${title} — ${company} — ${url}`);
      }
      say(`fetched ${Math.min(offset + 35, total)} / ${total} · ${matches.length} match(es)`);
      if (cfg.STOP_AT && matches.length >= cfg.STOP_AT) { say(`🎯 Reached ${cfg.STOP_AT} — stopping.`); break; }
      await delay(cfg.DELAY_MS);
    }

    if (!matches.length) { say("No matches found (try a different Skill, or clear Keywords)."); return; }

    console.table(matches.map(({ title, company, locations }) => ({ title, company, locations })));
    const q = s => `"${String(s).replace(/"/g, '""')}"`;
    const csv = "Title,Company,Locations,Skills,URL\n" +
      matches.map(m => [q(m.title), q(m.company), q(m.locations), q(m.skills), q(m.url)].join(",")).join("\n");
    const a = mk("a"); a.href = URL.createObjectURL(new Blob([csv], { type: "text/csv" }));
    a.download = `instahyre-jobs-${(cfg.SKILL || "search").replace(/\W+/g, "_").toLowerCase()}.csv`;
    document.body.appendChild(a); a.click(); a.remove();
    say(`✅ Done — ${matches.length} match(es). Saved ${a.download}.`);
    if (cfg.OPEN_TABS) for (const m of matches) { window.open(m.url, "_blank"); await delay(500); }
  }
})();
