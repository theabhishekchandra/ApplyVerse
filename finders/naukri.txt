/* ================================================================
   Naukri – Universal ApplyVerse  (interactive form)
   ----------------------------------------------------------------
   Paste this whole file into the DevTools Console while you're on a
   Naukri job-search results page (e.g. naukri.com/android-developer-
   jobs-in-india). A form panel appears — type your Keywords, Exclude
   words and options, then click ▶ Start. No code editing needed.

   HOW IT WORKS  (Naukri is different from ZipRecruiter/LinkedIn)
   • Naukri's job API is locked behind reCAPTCHA, so we DON'T call it.
   • Instead we read the jobs already rendered on the page, then click
     "Next" to advance. Naukri uses client-side routing, so the page
     changes WITHOUT a full reload — the script keeps running and
     scrapes each page in one go. No navigating you away, no CAPTCHA.
   • Each card already includes title, company, experience, location,
     salary, key-skills and a description snippet — we match your
     KEYWORDS against title + snippet + skills, drop EXCLUDE matches,
     de-dupe, show live progress, and download a CSV.

   HOW TO USE
   1. Search your role on Naukri (type it in Naukri's own search bar)
      and make sure you're on PAGE 1 of the results for full coverage.
   2. Console (Mac: Cmd+Option+J | Win: Ctrl+Shift+J) → paste → Enter.
   3. Fill the form → ▶ Start. CSV lands in Downloads when done.

   NOTES
   • It scrapes from the page you're currently on, forward. Start on
     page 1 to capture everything.
   • Change "Profile" only if you want a different role — it will
     navigate to that search (a real reload), after which you re-paste
     the script and Start again.
   • Keep DELAY sane; Naukri also watches for rapid automation.
   ================================================================ */

(function () {
  const $ = id => document.getElementById(id);
  const esc = s => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const delay = ms => new Promise(r => setTimeout(r, ms));
  const slugify = s => s.trim().toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");

  const mk = (tag, css, txt) => { const e = document.createElement(tag); if (css) e.style.cssText = css; if (txt != null) e.textContent = txt; return e; };
  function inputEl(id, val, ph) {
    const e = document.createElement("input");
    e.id = id; e.value = val || ""; e.placeholder = ph || "";
    e.style.cssText = "width:100%;box-sizing:border-box;padding:6px 8px;border:1px solid #3d5af1;border-radius:6px;background:#0d1633;color:#eaf0ff;font-size:12px";
    return e;
  }
  function fieldBlock(label, id, val, ph) {
    const w = mk("div");
    w.appendChild(mk("label", "display:block;margin:8px 0 3px;font-size:11px;opacity:.85", label));
    w.appendChild(inputEl(id, val, ph));
    return w;
  }
  function checkBlock(label, id, on) {
    const l = mk("label", "display:flex;align-items:center;gap:6px;margin:6px 0;font-size:12px;cursor:pointer");
    const c = document.createElement("input"); c.type = "checkbox"; c.id = id; c.checked = !!on;
    l.appendChild(c); l.appendChild(document.createTextNode(" " + label));
    return l;
  }

  // detect current search role from URL slug
  const slugMatch = location.pathname.match(/^\/(.+?)-jobs(?:-in-[a-z-]+)?(?:-\d+)?$/i);
  const curRole = slugMatch ? slugMatch[1].replace(/-/g, " ") : "";

  // ---------------- build form panel ----------------
  $("nkf-panel")?.remove();
  const p = mk("div", "position:fixed;top:16px;right:16px;width:340px;max-height:88vh;z-index:2147483647;"+
    "background:#0a1230;color:#eaf0ff;font:13px/1.45 -apple-system,Segoe UI,Roboto,sans-serif;"+
    "border:1px solid #3d5af1;border-radius:10px;box-shadow:0 8px 30px rgba(0,0,0,.5);display:flex;flex-direction:column;overflow:hidden");
  p.id = "nkf-panel";

  const header = mk("div", "padding:10px 12px;background:#1b2a6b;font-weight:600;display:flex;justify-content:space-between;align-items:center");
  header.appendChild(mk("span", "", "🔷 Naukri ApplyVerse"));
  const closeX = mk("span", "cursor:pointer;opacity:.7", "✕"); header.appendChild(closeX); p.appendChild(header);

  const form = mk("div", "padding:10px 12px;overflow:auto"); form.id = "nkf-form";
  form.appendChild(fieldBlock("Profile / search role (change = re-search)", "nkf-profile", curRole, "e.g. android developer"));
  form.appendChild(fieldBlock("Keywords (comma-sep — ANY match; blank = all)", "nkf-keywords", "android, kotlin, jetpack, flutter", "android, kotlin"));
  form.appendChild(fieldBlock("Exclude words (comma-sep — optional)", "nkf-exclude", "", "senior, sales, intern"));

  const two = mk("div", "display:flex;gap:8px");
  const c1 = mk("div", "flex:1"); c1.appendChild(fieldBlock("Max pages", "nkf-maxpages", "50"));
  const c2 = mk("div", "flex:1"); c2.appendChild(fieldBlock("Stop after N (0=all)", "nkf-stop", "0"));
  two.appendChild(c1); two.appendChild(c2); form.appendChild(two);

  const opts = mk("div", "margin-top:8px;padding-top:6px;border-top:1px solid #3d5af1");
  opts.appendChild(checkBlock("Remember seen jobs (only NEW next run)", "nkf-remember", false));
  opts.appendChild(checkBlock("Open each match in a new tab", "nkf-opentabs", false));
  form.appendChild(opts);

  const startBtn = mk("button", "width:100%;margin-top:12px;padding:9px;border:0;border-radius:7px;background:#4f7bff;color:#00102f;font-weight:700;font-size:13px;cursor:pointer", "▶ Start search");
  startBtn.id = "nkf-start"; form.appendChild(startBtn); p.appendChild(form);

  const status = mk("div", "padding:8px 12px;border-top:1px solid #3d5af1;font-size:12px;display:none"); status.id = "nkf-status";
  const list = mk("div", "overflow:auto;padding:2px 0"); list.id = "nkf-list";
  p.appendChild(status); p.appendChild(list);

  document.body.appendChild(p);
  closeX.onclick = () => p.remove();
  startBtn.onclick = start;

  const say = m => { status.textContent = m; console.log(m); };
  function addRow(r) {
    const row = mk("div", "padding:6px 12px;border-bottom:1px solid #1a2450");
    const a = mk("a", "color:#8fb4ff;text-decoration:none;font-weight:600;display:block", r.title);
    a.href = r.url; a.target = "_blank"; row.appendChild(a);
    row.appendChild(mk("div", "font-size:11px;opacity:.8", [r.company, r.location, r.exp].filter(Boolean).join(" · ")));
    list.appendChild(row); list.scrollTop = list.scrollHeight;
  }

  // robust to Naukri's layout variants: try known card containers, else derive from title links
  function findCards() {
    const sels = ["div.srp-jobtuple-wrapper", "div.cust-job-tuple", "article.jobTuple", "div.jobTuple"];
    for (const s of sels) { const els = document.querySelectorAll(s); if (els.length) return [...els]; }
    // last resort: group by job-title links and take their nearest card-ish ancestor
    const titles = [...document.querySelectorAll('a.title, a[href*="job-listings-"]')];
    return [...new Set(titles.map(t => t.closest("article, li, div.srp-jobtuple-wrapper, div.cust-job-tuple") || t.parentElement))].filter(Boolean);
  }
  const pick = (card, sels) => { for (const s of sels) { const el = card.querySelector(s); if (el && el.innerText.trim()) return el.innerText.trim(); } return ""; };
  const cleanUrl = u => (u || "").split("?")[0];   // strip tracking params so the same job isn't double-counted

  function scrapePage() {
    return findCards().map(card => {
      const tEl = card.querySelector('a.title') || card.querySelector('a[href*="job-listings-"]');
      const skills = [...card.querySelectorAll(".tag-li, ul.tags li")].map(t => t.innerText.trim()).filter(Boolean);
      return {
        title:   tEl?.innerText.trim() || "",
        url:     cleanUrl(tEl?.href) || "",
        company: pick(card, [".comp-name", "a.comp-name", ".subTitle", ".companyInfo"]),
        exp:     pick(card, [".expwdth", ".exp", ".experience"]),
        location:pick(card, [".locWdth", ".loc", ".location"]),
        salary:  pick(card, [".sal-wrap", ".sal", ".salary"]) || "—",
        snippet: pick(card, [".job-desc", ".job-description", ".jobDescription"]),
        skills:  skills.join(", "),
      };
    }).filter(j => j.title && j.url);
  }

  async function goNext() {
    const before = document.querySelector("div.srp-jobtuple-wrapper a.title")?.href || "";
    const next = [...document.querySelectorAll("a")].find(a => a.innerText.trim() === "Next" && a.href);
    if (!next || next.getAttribute("aria-disabled") === "true") return false;
    next.click();
    // wait for client-side route to swap the cards
    for (let i = 0; i < 30; i++) {
      await delay(300);
      const now = document.querySelector("div.srp-jobtuple-wrapper a.title")?.href || "";
      if (now && now !== before) return true;
    }
    return false; // no change => treat as end
  }

  async function start() {
    const splitList = s => s.split(",").map(x => x.trim()).filter(Boolean);
    const profile = $("nkf-profile").value.trim();
    // if the profile differs from the current search, navigate there (reload) and ask to re-run
    if (profile && slugify(profile) !== slugify(curRole)) {
      say(`↪ Switching search to "${profile}" — page will reload. Re-paste the script & Start again.`);
      await delay(1200);
      location.href = "https://www.naukri.com/" + slugify(profile) + "-jobs-in-india";
      return;
    }
    const cfg = {
      KEYWORDS: splitList($("nkf-keywords").value),
      EXCLUDE:  splitList($("nkf-exclude").value),
      MAX_PAGES:Math.max(1, parseInt($("nkf-maxpages").value) || 50),
      STOP_AT:  Math.max(0, parseInt($("nkf-stop").value)   || 0),
      REMEMBER: $("nkf-remember").checked,
      OPEN_TABS:$("nkf-opentabs").checked,
      DELAY_MS: 1200,
    };
    form.style.display = "none"; status.style.display = "block";
    await runSearch(cfg);
  }

  async function runSearch(cfg) {
    const kwRx = cfg.KEYWORDS.length ? new RegExp(cfg.KEYWORDS.map(esc).join("|"), "i") : null;
    const exRx = cfg.EXCLUDE.length  ? new RegExp(cfg.EXCLUDE.map(esc).join("|"),  "i") : null;
    const MEM_KEY = "nkf_seen_" + slugify(curRole || "all");
    const memSeen = new Set(cfg.REMEMBER ? JSON.parse(localStorage.getItem(MEM_KEY) || "[]") : []);

    say("Scanning Naukri results...");
    const seen = new Set(), matches = [];

    // guard: if the very first page has no cards, retry once (late render), then bail with a clear reason
    let firstJobs = scrapePage();
    if (!firstJobs.length) {
      say("No job cards yet — waiting for results to load...");
      await delay(2000);
      firstJobs = scrapePage();
      if (!firstJobs.length) {
        say("⚠ No job cards found. Open a Naukri SEARCH page (naukri.com/<role>-jobs-in-india) and let results load, then Start.");
        return;
      }
    }

    for (let page = 1; page <= cfg.MAX_PAGES; page++) {
      const jobs = page === 1 ? firstJobs : scrapePage();
      let hits = 0;
      for (const j of jobs) {
        if (seen.has(j.url)) continue;
        const text = [j.title, j.snippet, j.skills].join(" \n ");
        const keep = (!kwRx || kwRx.test(text)) && !(exRx && exRx.test(text));
        if (!keep) continue;
        seen.add(j.url);
        if (cfg.REMEMBER && memSeen.has(j.url)) continue;
        matches.push(j); addRow(j); hits++;
        console.log(`✓ ${j.title} — ${j.company} — ${j.url}`);
      }
      say(`page ${page}: ${jobs.length} jobs, ${hits} match. Total: ${matches.length}`);
      if (cfg.STOP_AT && matches.length >= cfg.STOP_AT) { say(`🎯 Reached ${cfg.STOP_AT} — stopping.`); break; }
      if (page >= cfg.MAX_PAGES) break;            // don't over-advance past the last page we'll scrape
      await delay(cfg.DELAY_MS);
      const advanced = await goNext();
      if (!advanced) { say(`📄 Page ${page} was the last page.`); break; }
    }

    if (!matches.length) { say("No matches found."); return; }
    if (cfg.REMEMBER) { matches.forEach(m => memSeen.add(m.url)); localStorage.setItem(MEM_KEY, JSON.stringify([...memSeen])); }

    console.table(matches.map(({ title, company, location, exp, salary }) => ({ title, company, location, exp, salary })));
    const q = s => `"${String(s).replace(/"/g, '""')}"`;
    const csv = "Title,Company,Location,Experience,Salary,Skills,URL\n" +
      matches.map(m => [q(m.title), q(m.company), q(m.location), q(m.exp), q(m.salary), q(m.skills), q(m.url)].join(",")).join("\n");
    const a = mk("a"); a.href = URL.createObjectURL(new Blob([csv], { type: "text/csv" }));
    a.download = `naukri-jobs-${slugify(curRole || "search")}.csv`;
    document.body.appendChild(a); a.click(); a.remove();
    say(`✅ Done — ${matches.length} match(es). Saved ${a.download}.`);
    if (cfg.OPEN_TABS) for (const m of matches) { window.open(m.url, "_blank"); await delay(500); }
  }
})();
